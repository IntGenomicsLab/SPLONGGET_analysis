Genotyping quantification comparison

Here we will show the scripts and commands used to plot the genotyping comparison results from the single-cell long-read data.

getVAF = function(vcf_file) {
  vcf = fread(vcf_file, skip = "#CHROM", header = TRUE)
  #vcf = vcf[grepl(paste0(targets$variants, collapse = "|"), paste0(vcf$`#CHROM`, ":", vcf$POS)), ]

  # Get the list of sample column names (assuming they start from the 10th column)

    vcf[, AD := as.numeric(gsub(";.*|AD=", "", INFO))]
    vcf[, DP := as.numeric(gsub(";OTH.*|.*DP=", "", INFO))]
    vcf[, VAF := AD / DP]

    return(data.table(variant_id = paste0(vcf[, `#CHROM`], ":", vcf[, POS]),
                                        VAF = vcf[, VAF],
                                        timepoint = basename(gsub("/cellSNP.*", "", vcf_file))))
}

res_cDNA = lapply(vcf_files_cDNA_targeted, getVAF) |> rbindlist()
res_cDNA[, type := "cDNA_targeted"]

res_DNA = lapply(vcf_files_DNA_targeted, getVAF) |> rbindlist()
res_DNA[, type := "DNA_targeted"]

res = rbind(res_cDNA, res_DNA)

# Merge with targets and remove CD19 and KRAS-C_T
res = merge(res, targets, by.x = "variant_id", by.y = "variants", all.x = TRUE)
res = res[!grepl("CD19|KRAS-C_T", variant_name)]

# Merge with missionbio VAFs
res = merge(res, missionbio_vafs, by = c("variant_name", "timepoint"), all.x = TRUE, suffixes = c("_SPLONGGET", "_missionbio"))

Plot UMAPs with genotyping results
