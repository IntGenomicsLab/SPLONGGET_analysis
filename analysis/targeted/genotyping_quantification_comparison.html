Define function to extract all relevant information from the vcf files.

# Function to extract information out vcf.
getGenotyping = function(vcf_file) {
  vcf = fread(vcf_file, skip = "#CHROM", header = TRUE)
  #vcf = vcf[grepl(paste0(targets$variants, collapse = "|"), paste0(vcf$`#CHROM`, ":", vcf$POS)), ]

  # Get the list of sample column names (assuming they start from the 10th column)
  sample_columns = colnames(vcf)[10:ncol(vcf)]

  # Initialize a list to store results
  mut_dt = pblapply(seq_len(nrow(vcf)), function(i) {
    # Extract the current row
    row_data = vcf[i, ..sample_columns]

    # Identify matching sample columns for genotypes '1/1', '0/1', '1/0'
    covered_mut = sample_columns[grepl("0/0", as.character(unlist(row_data)))]
    has_mut = sample_columns[grepl("1/1|0/1|1/0", as.character(unlist(row_data)))]

    dt = lapply(c(covered_mut, has_mut), function(x) {
      data.table(
        sample = x,
        variant_id = paste0(vcf[i, `#CHROM`], ":", vcf[i, POS]),
        genotype = gsub("^([^:]*):.*$", "\\1", vcf[i, ..x]),
        alt_depth = as.numeric(gsub("^([^:]*):([^:]*):.*$", "\\2", vcf[i, ..x])),
        total_depth = as.numeric(gsub("^([^:]*):([^:]*):([^:]*):.*$", "\\3", vcf[i, ..x])))
    }) |> rbindlist()
    return(dt)
  }, cl = 12)
  
  dt = rbindlist(mut_dt)
  dt[, timepoint := basename(gsub("/cellSNP.*", "", vcf_file))]
  dt[, barcode := paste0(sample, "-", timepoint)]
  return(dt)
}

# Get number of cells per timepoint

# Calculate percentage of power cells
res_power = res[power == "yes", .N, by = .(timepoint, variant_name, type)]
