Preprocess data

Then we get all the relevant barcodes and add metadata.

# Barcode matching
matched_barcodes = cbind(dna_barcodes, rna_barcodes)
setnames(matched_barcodes, c("atac_barcode", "cDNA_barcode"))

# Read in barcodes
untargeted_DNA = lapply(untargeted_DNA_files, function(x) {
    dt = fread(x)
    if (nrow(dt) == 0) {
        return(NULL)
    } else {
        dt[, timepoint := gsub("_.*", "", basename(x))]
        dt[, type := "untargeted"]
        dt[, modality := "DNA"]
        return(dt)
    }
}) |> rbindlist()

untargeted_cDNA = lapply(untargeted_cDNA_files, function(x) {
    dt = fread(x)
    if (nrow(dt) == 0) {
        return(NULL)
    } else {
        dt[, timepoint := gsub("_.*", "", basename(x))]
        dt[, type := "untargeted"]
        dt[, modality := "cDNA"]
        return(dt)
    }
}) |> rbindlist()

Combine with cell embeddings.

# Get UMAP
cell_embeddings = data.table(multiome@reductions$Xumap_@cell.embeddings)
cell_embeddings[, sample := rownames(multiome@reductions$Xumap_@cell.embeddings)]

# Adjust barcodes
untargeted_DNA = merge(untargeted_DNA, matched_barcodes, by.x = "V2", by.y = "atac_barcode", all.x = TRUE)
untargeted_DNA[, cDNA_barcode := paste0(cDNA_barcode, "-", timepoint)]
untargeted_DNA[, untargeted_dna_counts := V1]

untargeted_cDNA[, cDNA_barcode := paste0(V2, "-", timepoint)]
untargeted_cDNA[, untargeted_cdna_counts := V1]

total = merge(untargeted_DNA, untargeted_cDNA, by = "cDNA_barcode", all = TRUE)

# Set CART status
total[untargeted_dna_counts >= 1, untargeted_dna_CART_status := "CART"]
total[untargeted_cdna_counts >= 1, untargeted_cdna_CART_status := "CART"]
total[untargeted_dna_CART_status == "CART" & untargeted_cdna_CART_status == "CART", CART_status := "Detected in both"]
total[is.na(untargeted_dna_CART_status) & untargeted_cdna_CART_status == "CART", CART_status := "Detected in cDNA"]
total[untargeted_dna_CART_status == "CART" & is.na(untargeted_cdna_CART_status), CART_status := "Detected in DNA"]

# Merge with cell embeddings
total_umap = merge(cell_embeddings, total, by.y = "cDNA_barcode", by.x = "sample", all.x = TRUE)
total_umap[is.na(CART_status), CART_status := "Not detected"]

Plot UMAP

Finally we plot the UMAP with CART status.

ggplot(total_umap, aes(x = Xumap_1, y = Xumap_2, fill = CART_status)) +
    geom_point(data = total_umap[CART_status == "Not detected"], aes(x = Xumap_1, y = Xumap_2), size = 0.5, color = "lightgrey") +
    geom_point(data = total_umap[CART_status == "Detected in DNA"], aes(x = Xumap_1, y = Xumap_2), color = "#3994c9", size = 1) +
    geom_point(data = total_umap[CART_status == "Detected in cDNA"], aes(x = Xumap_1, y = Xumap_2), color = "#e7ac3d", size = 1) +
    geom_point(data = total_umap[CART_status == "Detected in both"], aes(x = Xumap_1, y = Xumap_2), color = "red", size = 1) +  
    scale_color_manual(values = c("#lightgrey", "#3994c9", "#e7ac3d", "red")) +
    labs(y = "UMAP 1", x = "UMAP 2", fill = "CAR T detection")
